<script>
const CONFIG = {
    brideName: "Kamola",
    groomName: "Kamoldin",

    partyDate: "2026-10-10",
    partyDateDisplay: "10-oktabr",

    weddingDate: "2026-10-11",
    weddingDateDisplay: "11-oktabr",

    weddingTime: "CHANGE_TIME",

    weddingAddress: "To‘raqo‘rg‘on tumani, Toshkent MFY, Kosonsoy ko‘chasi, 175-uy",

    landmark: "Odil Ahmedov yonidan kirganda",

    googleMapsLink: "CHANGE_GOOGLE_MAPS_LINK",

    weddingMusicUrl: "CHANGE_MUSIC_URL"
};

const $ = s => document.querySelector(s);
const V = { ...CONFIG, couple: CONFIG.groomName + " & " + CONFIG.brideName };
document.querySelectorAll("[data-cfg]").forEach(el => el.textContent = V[el.dataset.cfg]);
document.title = V.couple + " — To‘y taklifnomasi";
const reduce = matchMedia("(prefers-reduced-motion: reduce)").matches;

// Loader
addEventListener("load", () => setTimeout(() => $("#loader").classList.add("gone"), 1100));

// Music
const audio = new Audio();
audio.loop = true; audio.preload = "auto";
const musicOk = CONFIG.weddingMusicUrl && !CONFIG.weddingMusicUrl.startsWith("CHANGE_");
if (musicOk) audio.src = CONFIG.weddingMusicUrl;
const mBtn = $("#music");
const mLabel = () => mBtn.textContent = audio.paused ? "🔇 Paused" : "🎵 Playing";
audio.addEventListener("play", mLabel); audio.addEventListener("pause", mLabel);
mBtn.onclick = () => audio.paused ? audio.play().catch(mLabel) : audio.pause();

// Open invitation (this click starts the music)
$("#openBtn").onclick = () => {
    if (musicOk) { audio.play().catch(mLabel); mBtn.hidden = false; mLabel(); }
    $("#opening").classList.add("hide");
    document.body.classList.remove("lock");
    $("#main").classList.add("show");
    scrollTo(0, 0);
    if (!reduce) petals();
};

// Countdown
const target = new Date(CONFIG.weddingDate + "T00:00:00");
function tick() {
    const ms = target - Date.now();
    if (ms <= 0) { $("#count").hidden = true; $("#cdDone").hidden = false; return; }
    const t = Math.floor(ms / 1000);
    $("#cd").textContent = Math.floor(t / 86400);
    $("#ch").textContent = Math.floor(t % 86400 / 3600);
    $("#cm").textContent = Math.floor(t % 3600 / 60);
    $("#cs").textContent = t % 60;
    setTimeout(tick, 1000 - Date.now() % 1000);
}
tick();

// Maps
$("#mapBtn").onclick = () => {
    if (CONFIG.googleMapsLink.startsWith("CHANGE_")) return alert("Google Maps havolasi hali kiritilmagan.");
    window.open(CONFIG.googleMapsLink, "_blank", "noopener");
};

// RSVP
const msg = $("#rsvpMsg"), guest = $("#guest");
document.querySelectorAll("[data-rsvp]").forEach(b => b.onclick = () => {
    const name = guest.value.trim();
    guest.classList.toggle("err", !name);
    if (!name) { msg.textContent = "Iltimos, ismingizni kiriting."; return guest.focus(); }
    msg.textContent = b.dataset.rsvp === "yes" ? "Rahmat, " + name + "! Sizni to‘yimizda kutamiz ❤️" : "Rahmat, " + name + ". Javobingiz uchun minnatdormiz.";
    msg.classList.remove("show"); void msg.offsetWidth; msg.classList.add("show");
});

// Scroll reveal
const io = new IntersectionObserver(es => es.forEach(e => { if (e.isIntersecting) { e.target.classList.add("in"); io.unobserve(e.target); } }), { threshold: .15 });
document.querySelectorAll(".rv").forEach((el, i) => { if (el.closest(".tl")) el.style.setProperty("--d", "0.05s"); io.observe(el); });

// Progress + back to top
const bar = $("#progress"), top = $("#top");
let raf;
addEventListener("scroll", () => raf = raf || requestAnimationFrame(() => {
    raf = 0;
    const h = document.documentElement.scrollHeight - innerHeight;
    bar.style.width = (h > 0 ? scrollY / h * 100 : 0) + "%";
    top.classList.toggle("on", scrollY > 400);
}), { passive: true });
top.onclick = () => scrollTo({ top: 0, behavior: reduce ? "auto" : "smooth" });

// Petals (few, subtle)
function petals() {
    ["🌸", "💗", "🌷", "✨", "🌸", "💗", "🌷", "✨"].forEach((c, i) => {
        const p = document.createElement("span");
        p.className = "petal"; p.textContent = c;
        p.style.cssText = `left:${Math.random() * 95}%;font-size:${14 + Math.random() * 12}px;animation-duration:${14 + Math.random() * 10}s;animation-delay:${-i * 3}s`;
        document.body.appendChild(p);
    });
}
